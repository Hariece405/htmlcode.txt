<script>

    const seatsContainer = document.getElementById("seats");

    let selectedSeats = [];

    // Create 50 seats
    for (let i = 1; i <= 50; i++) {

        const seat = document.createElement("div");

        seat.classList.add("seat");

        seat.innerText = i;

        // Randomly mark some seats as booked
        if ([3, 7, 12, 18, 25, 31, 42].includes(i)) {
            seat.classList.add("booked");
        }

        seat.addEventListener("click", function () {

            if (seat.classList.contains("booked")) {
                return;
            }

            if (seat.classList.contains("selected")) {

                seat.classList.remove("selected");

                selectedSeats = selectedSeats.filter(
                    s => s !== i
                );

            } else {

                seat.classList.add("selected");

                selectedSeats.push(i);
            }

            updateSummary();
        });

        seatsContainer.appendChild(seat);
    }


    function selectTime(button) {

        document
            .querySelectorAll(".time-btn")
            .forEach(btn => btn.classList.remove("selected"));

        button.classList.add("selected");
    }


    function updatePrice() {

        updateSummary();
    }


    function updateSummary() {

        const price = Number(
            document.getElementById("movie").value
        );

        const total = price * selectedSeats.length;

        document.getElementById("total").innerText = total;

        document.getElementById("selectedSeats").innerText =
            selectedSeats.length
                ? selectedSeats.join(", ")
                : "None";
    }


    function bookTickets() {

        const movie =
            document.getElementById("movie")
            .options[
                document.getElementById("movie").selectedIndex
            ].text;

        const date =
            document.getElementById("date").value;

        const selectedTime =
            document.querySelector(".time-btn.selected");

        if (selectedSeats.length === 0) {

            alert("Please select at least one seat.");

            return;
        }

        if (!selectedTime) {

            alert("Please select a show time.");

            return;
        }

        const message =
            document.getElementById("message");

        message.style.display = "block";

        message.innerHTML = `
            <strong>🎉 Booking Confirmed!</strong><br><br>

            Movie: ${movie}<br>
            Date: ${date}<br>
            Time: ${selectedTime.innerText}<br>
            Seats: ${selectedSeats.join(", ")}<br>
            Total: ₹${document.getElementById("total").innerText}
        `;
    }

</script>
